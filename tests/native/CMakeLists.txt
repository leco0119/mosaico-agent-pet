cmake_minimum_required(VERSION 3.20)
project(pet_usage_tests LANGUAGES C)
enable_testing()
set(PET "${CMAKE_CURRENT_LIST_DIR}/../../components/app_pet")
set(USAGE_LINK "${CMAKE_CURRENT_LIST_DIR}/../../components/app_usage_link")
add_executable(test_usage_payload "${PET}/usage/usage_payload.c" test_usage_payload.c)
target_include_directories(test_usage_payload PRIVATE "${PET}/include")
add_test(NAME usage_payload COMMAND test_usage_payload)
add_executable(test_usage_store "${PET}/usage/usage_payload.c" "${USAGE_LINK}/usage_store.c"
    test_usage_store.c stubs/fake_nvs.c)
target_include_directories(test_usage_store PRIVATE stubs "${PET}/include" "${USAGE_LINK}")
add_test(NAME usage_store COMMAND test_usage_store)
add_executable(test_usage_controller "${PET}/usage/app_usage.c" "${PET}/usage/usage_payload.c"
    test_usage_controller.c)
target_include_directories(test_usage_controller PRIVATE stubs "${PET}/include")
add_test(NAME usage_controller COMMAND test_usage_controller)
add_executable(test_pet_pages "${PET}/pet_pages.c" "${PET}/usage/app_usage.c"
    "${PET}/usage/usage_payload.c" test_pet_pages.c)
target_include_directories(test_pet_pages PRIVATE stubs "${PET}/include"
    "${PET}/../app_voice/include" "${PET}/../app_wifi/include")
add_test(NAME pet_pages COMMAND test_pet_pages)
foreach(target test_usage_payload test_usage_store test_usage_controller test_pet_pages)
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8 /UNDEBUG /W4)
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_options(${target} PRIVATE -UNDEBUG -Wall -Wextra -Werror)
    endif()
endforeach()
